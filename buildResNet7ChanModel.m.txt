function lgraph = buildResNet7ChanModel(classNames, classWeights)
net = resnet50;
lgraph = layerGraph(net);

% ---- 1. TRUE freeze for backbone; only res4/res5 fine-tune ----
for i = 1:numel(lgraph.Layers)
    layer = lgraph.Layers(i);
    if isprop(layer, 'WeightLearnRateFactor')
        if startsWith(layer.Name, 'res5') || startsWith(layer.Name, 'bn5') || ...
           startsWith(layer.Name, 'res4') || startsWith(layer.Name, 'bn4')
            layer.WeightLearnRateFactor = 0.5;
            layer.BiasLearnRateFactor   = 0.5;
        else
            layer.WeightLearnRateFactor = 0;   % TRUE freeze, was 0.1
            layer.BiasLearnRateFactor   = 0;
        end
        lgraph = replaceLayer(lgraph, layer.Name, layer);
    end
end

% ---- Rebuild conv1 for 7-channel input ----
oldConv1 = net.Layers(2);
oldWeights = oldConv1.Weights;
oldBias    = oldConv1.Bias;
numFilters = size(oldWeights, 4);
newWeights = zeros(7, 7, 7, numFilters, 'single');
newWeights(:,:,1:3,:) = oldWeights;
newWeights(:,:,4:7,:) = single(0.01 * randn(7,7,4,numFilters));  % smaller init, was 0.1

newConv1 = convolution2dLayer(7, numFilters, ...
    'Stride', 2, 'Padding', [3 3 3 3], ...
    'Name', 'conv1', ...
    'Weights', newWeights, 'Bias', oldBias, ...
    'WeightLearnRateFactor', 1, 'BiasLearnRateFactor', 1);   % reduced from 10

lgraph = replaceLayer(lgraph, 'conv1', newConv1);

% ---- Input layer ----
newInput = imageInputLayer([224 224 7], ...
    'Name', 'input_7chan', ...
    'Normalization', 'none');
lgraph = replaceLayer(lgraph, 'input_1', newInput);

% ---- New 5-class head with dropout to fight overfitting ----
lgraph = removeLayers(lgraph, {'fc1000', 'fc1000_softmax', 'ClassificationLayer_fc1000'});
numClasses = numel(classNames);

newHead = [
    dropoutLayer(0.4, 'Name','dropout_dr')
    fullyConnectedLayer(numClasses, 'Name', 'fc5_dr', ...
        'WeightLearnRateFactor', 1, 'BiasLearnRateFactor', 1)   % reduced from 10
    softmaxLayer('Name', 'softmax_dr')
    classificationLayer('Name', 'classoutput_dr', ...
        'Classes', classNames, 'ClassWeights', classWeights)
];
lgraph = addLayers(lgraph, newHead);
lgraph = connectLayers(lgraph, 'avg_pool', 'dropout_dr');
end
